-- Create admin_logs table if it doesn't exist
-- This table is referenced by multiple admin endpoints but was missing from the codebase
--
-- Run this SQL in your database to create the table:
-- psql $DATABASE_URL -f create_admin_logs.txt

CREATE TABLE IF NOT EXISTS admin_logs (
    id SERIAL PRIMARY KEY,
    admin_id INTEGER NOT NULL REFERENCES auth_users(id) ON DELETE SET NULL,
    action_type VARCHAR(50) NOT NULL,
    target_user_id INTEGER REFERENCES auth_users(id) ON DELETE SET NULL,
    target_entity_type VARCHAR(50),
    target_entity_id UUID,
    details JSONB DEFAULT '{}',
    created_at TIMESTAMP DEFAULT NOW()
);

-- Create indexes for common queries
CREATE INDEX IF NOT EXISTS idx_admin_logs_admin_id ON admin_logs(admin_id);
CREATE INDEX IF NOT EXISTS idx_admin_logs_target_user ON admin_logs(target_user_id);
CREATE INDEX IF NOT EXISTS idx_admin_logs_action_type ON admin_logs(action_type);
CREATE INDEX IF NOT EXISTS idx_admin_logs_created_at ON admin_logs(created_at DESC);

COMMENT ON TABLE admin_logs IS 'Audit trail for administrative actions';
COMMENT ON COLUMN admin_logs.action_type IS 'Type of action: verification_toggle, user_promote, report_action, etc.';
COMMENT ON COLUMN admin_logs.details IS 'JSON object with action-specific context';
